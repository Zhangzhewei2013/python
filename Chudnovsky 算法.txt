from decimal import Decimal, getcontext

def compute_pi(digits: int) -> Decimal:
    """
    计算圆周率，保留 digits 位小数。
    digits 表示小数点后的位数。
    """
    # 多设置一些精度，避免中间计算误差
    getcontext().prec = digits + 20

    C = 426880 * Decimal(10005).sqrt()
    M = 1
    L = 13591409
    X = 1
    K = 6
    S = Decimal(L)

    # 每项约贡献 14 位精度
    for i in range(1, digits // 14 + 2):
        M = (M * (K**3 - 16 * K)) // (i**3)
        L += 545140134
        X *= -262537412640768000
        S += Decimal(M * L) / X
        K += 12

    pi = C / S
    return +pi

if __name__ == "__main__":
    n = 1000  # 计算 1000 位小数
    pi = compute_pi(n)
    print(format(pi, f".{n}f"))
