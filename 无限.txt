from decimal import Decimal, getcontext
import sys
import time

def compute_pi(digits: int) -> Decimal:
    """计算圆周率，保留 digits 位小数（固定精度算法）"""
    getcontext().prec = digits + 20

    C = 426880 * Decimal(10005).sqrt()
    M = 1
    L = 13591409
    X = 1
    K = 6
    S = Decimal(L)

    for i in range(1, digits // 14 + 2):
        M = (M * (K**3 - 16 * K)) // (i**3)
        L += 545140134
        X *= -262537412640768000
        S += Decimal(M * L) / X
        K += 12

    return +(C / S)


if __name__ == "__main__":
    step = 100          # 每次增加的位数，可调大/调小
    digits = step       # 当前目标小数位数
    printed = 0         # 已经输出的小数位数

    # 先输出整数部分和小数点
    print("3.", end="", flush=True)

    try:
        while True:
            t0 = time.time()
            pi = compute_pi(digits)
            pi_str = format(pi, f".{digits}f")   # 形如 "3.1415..."
            frac = pi_str[2:]                    # 去掉 "3."

            # 只输出新增的部分
            if len(frac) > printed:
                sys.stdout.write(frac[printed:])
                sys.stdout.flush()
                printed = len(frac)

            cost = time.time() - t0
            # 每 1000 位换行并提示一次（不影响小数流）
            if printed % 1000 < step:
                print(f"\n[已输出 {printed} 位，本轮耗时 {cost:.2f}s]", file=sys.stderr)

            digits += step

    except KeyboardInterrupt:
        print(f"\n\n已停止，共输出 {printed} 位小数。")
